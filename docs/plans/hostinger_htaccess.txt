# ImportFlow update host: .htaccess for public_html/importflow/updates/ (plan docs/plans/update-system-plan.md section 9, step 4)
#
# Upload this file as:  public_html/importflow/updates/.htaccess   (hPanel File Manager or SFTP on port 65002)
# Everything from the "Options" line down is the file content; these comment lines are harmless to keep.
#
# - No directory listings.
# - In-flight uploads (*.part installer, manifest.json.tmp) are never served.
# - manifest.json is never cached (the app also adds ?t=<timestamp>); installers are immutable per version path.
# - If the Hostinger CDN is on, also add a CDN bypass rule for /importflow/updates/manifest.json.
# - The Header lines need mod_headers (LiteSpeed on Hostinger supports it). The IfModule guard stops a missing
#   module from turning the whole folder into HTTP 500.

Options -Indexes
AddType application/json .json
AddType application/octet-stream .exe

<FilesMatch "\.(part|tmp)$">
  Require all denied
</FilesMatch>

<IfModule mod_headers.c>
  <Files "manifest.json">
    Header set Cache-Control "no-store, max-age=0"
  </Files>
  <FilesMatch "\.exe$">
    Header set Cache-Control "public, max-age=31536000, immutable"
    Header set X-Content-Type-Options "nosniff"
  </FilesMatch>
</IfModule>
